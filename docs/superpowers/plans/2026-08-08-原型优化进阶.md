  a.shelveUntil=ms?Date.now()+ms:Infinity;

function isShelved(a){return a.shelveUntil!=null&&a.shelveUntil>Date.now();}

- [ ] **Step 3: `addAlarm` 与 `renderAlarms` 感知屏蔽**

在 `addAlarm` 的入队处（`alarms.unshift(...)` 前）跳过仍被屏蔽的同源告警；在 `renderAlarms` 的每条渲染中，被屏蔽的项显示「已屏蔽」标签 + 「解除」按钮，且 `updAlarm` 的未确认计数排除已屏蔽项：

```javascript
function addAlarm(lv,msg){
  playAlarm(lv);
  const now=new Date().toLocaleTimeString('zh-CN',{hour12:false});
  const dup=alarms.find(a=>a.msg===msg);
  if(dup){dup.time=now;dup.acked=false;saveAlarms();renderAlarms();$('sb-msg').textContent='⚠ '+msg;return;}
  if(alarms.some(a=>a.msg===msg&&isShelved(a)))return; // 屏蔽期间不重复入队
  alarms.unshift({lv,msg,time:now,acked:false,shelveUntil:0});
  if(alarms.length>200)alarms.pop();
  saveAlarms();renderAlarms();updAlarm();$('sb-msg').textContent='⚠ '+msg;
}
function updAlarm(){const n=alarms.filter(a=>!a.acked&&!isShelved(a)).length;const b=$('sb-alarm');b.textContent=n;$('sb-alarm-wrap').classList.toggle('has',n>0);}
```

在 `renderAlarms` 的模板字符串中追加屏蔽操作列：

```javascript
const sheld=isShelved(a);
el.innerHTML=`${warnSvg}<span class="lv">${a.lv}</span><span class="msg">${a.msg}</span><span class="time">${a.time}</span>${sheld?'<span class="ack">已屏蔽</span>':'<button class="ack" data-shelve>屏蔽 1h</button>'}<button class="ack" data-unshelve>${sheld?'解除':'确认'}</button>`;
if(!a.acked&&!sheld)el.querySelector('[data-shelve]').onclick=()=>shelveAlarm(a,3600e3);

else el.querySelector('[data-unshelve]').onclick=()=>{a.acked=true;saveAlarms();renderAlarms();updAlarm();};
```
